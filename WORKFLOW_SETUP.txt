name: Update news and publish
on:
  push:
    branches: [main]
  workflow_dispatch:
  schedule:
    - cron: '17 */6 * * *'
permissions:
  contents: read
  pages: write
  id-token: write
concurrency:
  group: github-pages
  cancel-in-progress: false
jobs:
  publish:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v6
      - uses: actions/configure-pages@v5
      - name: Refresh publisher feeds
        run: python3 update-news.py
      - name: Prepare website files
        run: |
          mkdir -p _site
          cp index.html app.js style.css news.json _site/
          if [ -f app.js.LEGAL.txt ]; then cp app.js.LEGAL.txt _site/; fi
          touch _site/.nojekyll
      - uses: actions/upload-pages-artifact@v4
        with:
          path: _site
      - name: Publish
        id: deployment
        uses: actions/deploy-pages@v4
